using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class EvolveAbility : Ability
{

	public CombatAction actionWhenInAttackMode {get; private set;}

	public override Stats actorStats
	{
		set
		{
			base.actorStats = value;
			actionWhenInAttackMode.actorStats = value;

			// if (value != null)
			// {
			// 	Debug.LogError("actorStats set to " + value.position.ToString());
			// }
		}
	}

	public EvolveAbility(CombatActionSettings settings, CombatAction actionWhenInAttackMode, List<TargetingLogic> targetingLogics = null):
	base(settings, targetingLogics)
	{
		this.actionWhenInAttackMode = actionWhenInAttackMode.clone();
	}

	public override SelectorTemplate getRangeTemplate()
	{
		if(inAttackMode())
		{
			return actionWhenInAttackMode.getRangeTemplate();
		} else
		{
			return base.getRangeTemplate();
		}
	}

	public override string getRangeTitle()
	{
		if(inAttackMode())
		{
			return actionWhenInAttackMode.getRangeTemplate().ToFriendlyString();
		} else
		{
			return base.getRangeTemplate().ToFriendlyString();
		}
	}
	
	public override string displayName
	{
		get
		{
			if(inAttackMode())
			{
				return actionWhenInAttackMode.displayName;
			} else
			{
				return base.displayName;
			}
		}
	}

	public override string uniqueName
	{
		get
		{
			if(inAttackMode())
			{
				return actionWhenInAttackMode.uniqueName;
			} else
			{
				return base.uniqueName;
			}
		}
	}
	
	public override string iconName
	{
		get
		{
			if(inAttackMode())
			{
				return actionWhenInAttackMode.iconName;
			} else
			{
				return base.iconName;
			}
		}
	}
	
	public override string getUseDescription()
	{
		if(inAttackMode())
		{
			return actionWhenInAttackMode.getUseDescription();
		} else
		{
			return base.getUseDescription();
		}
	}

	public static bool inAttackMode()
	{
		return CombatStateManager.turnNumber % 2 == 1;
	}
	
	public override Selector selector
	{
		set
		{
			base.selector = value;
			actionWhenInAttackMode.selector = value;
		}
	}

	
	/*public override void setTargetCoords(GridCoords newTargetCoords)
	{
		base.setTargetCoords(newTargetCoords);
		actionWhenInAttackMode.setTargetCoords(newTargetCoords);
	}
	*/

}
