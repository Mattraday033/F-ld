using System;
using System.Collections;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;

public class TakeHostageAbility : Ability
{

    public TakeHostageAbility(CombatActionSettings settings, List<TargetingLogic> targetingLogics = null) :
    base(settings, targetingLogics)
    {

    }

    public override int findFinalDamage(Stats target, bool isCrit)
    {
        int finalDamage = 0;

        if(target != null)
        {
            finalDamage = target.currentHealth * 3;
        }

        return finalDamage;
    }


}
